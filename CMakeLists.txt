cmake_minimum_required(VERSION 3.16)
project(arcsound VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(ARCSOUND_SHARED "build libarcsound as a shared library" OFF)
option(ARCSOUND_SANITIZE "build with AddressSanitizer + UndefinedBehaviorSanitizer (gcc / clang)" OFF)
if(ARCSOUND_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()

set(ARCSOUND_SRC src/arcsound.c third_party/cjson/cJSON.c)
if(ARCSOUND_SHARED)
  add_library(arcsound SHARED ${ARCSOUND_SRC})
else()
  add_library(arcsound STATIC ${ARCSOUND_SRC})
endif()
target_include_directories(arcsound PUBLIC include)
if(NOT MSVC)
  target_compile_options(arcsound PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(arcsound PUBLIC m)
endif()

add_executable(arcsound_cli tools/arcsound_cli.c)
target_link_libraries(arcsound_cli arcsound)

enable_testing()
add_executable(test_stream tests/test_stream.c)
target_link_libraries(test_stream arcsound)
add_test(NAME stream_chunking COMMAND test_stream ${CMAKE_CURRENT_SOURCE_DIR}/model)
set_tests_properties(stream_chunking PROPERTIES SKIP_RETURN_CODE 77)
